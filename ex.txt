#include <stdio.h>
#include <mpi.h>

#define N 16

int main(int argc, char *argv[]) {
    int rank, size, a[N], local[N];
    int sum = 0, total = 0, seq = 0;
    double t, seq_time = 0, mpi_time;

    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size > N || N % size != 0) {
        if (rank == 0)
            printf("Invalid process count\n");
        MPI_Finalize();
        return 1;
    }

    if (rank == 0) {
        for (int i = 0; i < N; i++)
            a[i] = i + 1;

        t = MPI_Wtime();
        for (int i = 0; i < N; i++)
            seq += a[i];
        seq_time = MPI_Wtime() - t;
    }

    MPI_Barrier(MPI_COMM_WORLD);
    t = MPI_Wtime();

    MPI_Scatter(a, N / size, MPI_INT, local,
                N / size, MPI_INT, 0, MPI_COMM_WORLD);

    for (int i = 0; i < N / size; i++)
        sum += local[i];

    MPI_Reduce(&sum, &total, 1, MPI_INT,
               MPI_SUM, 0, MPI_COMM_WORLD);

    mpi_time = MPI_Wtime() - t;

    if (rank == 0) {
        printf("Sequential sum = %d\n", seq);
        printf("MPI sum = %d\n", total);
        printf("Sequential time = %.8f seconds\n", seq_time);
        printf("MPI time = %.8f seconds\n", mpi_time);

        if (mpi_time > 0)
            printf("Speedup = %.4f\n", seq_time / mpi_time);
    }

    MPI_Finalize();
    return 0;
}

#include <stdio.h>
#include <mpi.h>

#define N 16

int main(int argc, char *argv[]) {
    int a[N], b[N], rank, size, local = 0, max = 0, seq = 0;
    double t, st, pt;

    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size > N || N % size != 0) {
        if (rank == 0) printf("Invalid process count\n");
        MPI_Finalize();
        return 1;
    }

    if (rank == 0) {
        for (int i = 0; i < N; i++) a[i] = i + 1;
        t = MPI_Wtime();
        for (int i = 0; i < N; i++)
            if (a[i] > seq) seq = a[i];
        st = MPI_Wtime() - t;
    }

    MPI_Barrier(MPI_COMM_WORLD);
    t = MPI_Wtime();

    MPI_Scatter(a, N / size, MPI_INT, b, N / size,
                MPI_INT, 0, MPI_COMM_WORLD);

    for (int i = 0; i < N / size; i++)
        if (b[i] > local) local = b[i];

    MPI_Reduce(&local, &max, 1, MPI_INT, MPI_MAX,
               0, MPI_COMM_WORLD);

    pt = MPI_Wtime() - t;

    if (rank == 0) {
        printf("Sequential maximum = %d\n", seq);
        printf("MPI maximum = %d\n", max);
        printf("Sequential time = %f\n", st);
        printf("MPI time = %f\n", pt);
        if (pt > 0) printf("Speedup = %f\n", st / pt);
    }

    MPI_Finalize();
    return 0;
}

#include <stdio.h>
#include <mpi.h>

#define N 16

int main(int argc, char *argv[]) {
    int a[N], b[N], rank, size;
    long long local = 1, prod = 1, seq = 1;
    double t, st, pt;

    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size > N || N % size != 0) {
        if (rank == 0) printf("Invalid process count\n");
        MPI_Finalize();
        return 1;
    }

    if (rank == 0) {
        for (int i = 0; i < N; i++)
            a[i] = (i % 2 == 0) ? 1 : 2;

        t = MPI_Wtime();
        for (int i = 0; i < N; i++) seq *= a[i];
        st = MPI_Wtime() - t;
    }

    MPI_Barrier(MPI_COMM_WORLD);
    t = MPI_Wtime();

    MPI_Scatter(a, N / size, MPI_INT, b, N / size,
                MPI_INT, 0, MPI_COMM_WORLD);

    for (int i = 0; i < N / size; i++) local *= b[i];

    MPI_Reduce(&local, &prod, 1, MPI_LONG_LONG_INT,
               MPI_PROD, 0, MPI_COMM_WORLD);

    pt = MPI_Wtime() - t;

    if (rank == 0) {
        printf("Sequential product = %lld\n", seq);
        printf("MPI product = %lld\n", prod);
        printf("Sequential time = %f\n", st);
        printf("MPI time = %f\n", pt);
        if (pt > 0) printf("Speedup = %f\n", st / pt);
    }

    MPI_Finalize();
    return 0;
}
Matrix multiplication using OpenMP==========================================================================

#include <stdio.h>
#include <omp.h>
#define N 200

int main() {
    static int a[N][N], b[N][N], c[N][N], d[N][N];
    int r1 = N, c1 = N, r2 = N, c2 = N;
    int threads[] = {2, 4, 8};
    double t;

    if (c1 != r2) {
        printf("Matrix dimension mismatch\n");
        return 1;
    }

    for (int i = 0; i < N; i++)
        for (int j = 0; j < N; j++) {
            a[i][j] = 1;
            b[i][j] = 2;
        }

    t = omp_get_wtime();
    for (int i = 0; i < r1; i++)
        for (int j = 0; j < c2; j++)
            for (int k = 0; k < c1; k++)
                c[i][j] += a[i][k] * b[k][j];

    printf("Sequential time = %f\n", omp_get_wtime() - t);

    for (int x = 0; x < 3; x++) {
        int th = threads[x];
        if (th < 1 || th > omp_get_num_procs()) {
            printf("Invalid thread count: %d\n", th);
            continue;
        }

        t = omp_get_wtime();

        #pragma omp parallel for num_threads(th)
        for (int i = 0; i < r1; i++)
            for (int j = 0; j < c2; j++)
                for (int k = 0; k < c1; k++)
                    d[i][j] += a[i][k] * b[k][j];

        printf("Threads = %d, Result = %d, Time = %f\n",
               th, d[0][0], omp_get_wtime() - t);
    }
    return 0;
}

#include <stdio.h>
#include <omp.h>
#define N 1000000

int main() {
    static int a[N];
    int mn = 0, pmn;
    double t;
    int threads[] = {2, 4, 8};

    for (int i = 0; i < N; i++) a[i] = N - i;

    t = omp_get_wtime();
    mn = a[0];
    for (int i = 1; i < N; i++)
        if (a[i] < mn) mn = a[i];

    printf("Sequential min = %d, Time = %f\n",
           mn, omp_get_wtime() - t);

    for (int k = 0; k < 3; k++) {
        int th = threads[k];
        if (th < 1 || th > omp_get_num_procs()) {
            printf("Invalid thread count: %d\n", th);
            continue;
        }

        pmn = a[0];
        t = omp_get_wtime();

        #pragma omp parallel for num_threads(th) reduction(min:pmn)
        for (int i = 0; i < N; i++)
            if (a[i] < pmn) pmn = a[i];

        printf("Threads = %d, Min = %d, Time = %f\n",
               th, pmn, omp_get_wtime() - t);
    }
    return 0;
}

#include <stdio.h>
#include <omp.h>
#define N 1000000

int main() {
    static int a[N];
    long long sum = 0, psum = 0;
    double t;
    int threads[] = {2, 4, 8};

    for (int i = 0; i < N; i++) a[i] = 1;

    t = omp_get_wtime();
    for (int i = 0; i < N; i++) sum += a[i];
    printf("Sequential sum = %lld, Time = %f\n",
           sum, omp_get_wtime() - t);

    for (int k = 0; k < 3; k++) {
        int th = threads[k];
        if (th < 1 || th > omp_get_num_procs()) {
            printf("Invalid thread count: %d\n", th);
            continue;
        }

        psum = 0;
        t = omp_get_wtime();

        #pragma omp parallel for num_threads(th) reduction(+:psum)
        for (int i = 0; i < N; i++) psum += a[i];

        printf("Threads = %d, Sum = %lld, Time = %f\n",
               th, psum, omp_get_wtime() - t);
    }
    return 0;
}

def solve(n, row=0, board=[]):
    if row == n:
        print(board)
        return True
    for col in range(n):
        if all(col != c and abs(col-c) != row-r
               for r, c in enumerate(board)):
            if solve(n, row+1, board+[col]):
                return True
    return False

    if i == len(a) or target < 0:

import heapq

def knapsack(w, v, cap):
    n = len(w)
    items = sorted(zip(w, v), key=lambda x: x[1]/x[0], reverse=True)

    def bound(i, weight, value):
        b, rem = value, cap-weight
        while i < n and items[i][0] <= rem:
            rem -= items[i][0]
            b += items[i][1]
            i += 1
        if i < n:
            b += items[i][1] * rem / items[i][0]
        return b

    pq = [(-bound(0, 0, 0), 0, 0, 0)]
    best = 0

    while pq:
        negb, i, weight, value = heapq.heappop(pq)
        if -negb <= best or i == n:
            continue

        wi, vi = items[i]
        if weight + wi <= cap:
            nw, nv = weight + wi, value + vi
            best = max(best, nv)
            if i+1 < n and bound(i+1, nw, nv) > best:
                heapq.heappush(pq, (-bound(i+1, nw, nv), i+1, nw, nv))

        if i+1 < n and bound(i+1, weight, value) > best:
            heapq.heappush(pq, (-bound(i+1, weight, value), i+1, weight, value))

    return best

print(knapsack([2, 3, 4, 5], [3, 4, 5, 6], 5))

def bubble(a):
    n = len(a)
    for i in range(n):
        swapped = False
        for j in range(n-i-1):
            if a[j] > a[j+1]:
                a[j], a[j+1] = a[j+1], a[j]
                swapped = True
        if not swapped:
            break
    return a

def linear(a, x):
    for i, v in enumerate(a):
        if v == x:
            return i
    return -1

a = [5, 2, 8, 1, 3]
print(bubble(a))
print(linear(a, 8))

def quick(a):
    if len(a) <= 1:
        return a
    p = a[len(a)//2]
    return (quick([x for x in a if x < p]) +
            [x for x in a if x == p] +
            quick([x for x in a if x > p]))

def binary(a, x):
    l, r = 0, len(a)-1
    while l <= r:
        m = (l+r)//2
        if a[m] == x:
            return m
        if a[m] < x:
            l = m+1
        else:
            r = m-1
    return -1

def merge_sort(a):
    if len(a) <= 1:
        return a
    m = len(a)//2
    l, r = merge_sort(a[:m]), merge_sort(a[m:])
    out = []
    while l and r:
        out.append((l if l[0] <= r[0] else r).pop(0))
    return out + l + r

def prim(n, edges):
    g = [[] for _ in range(n)]
    for u, v, w in edges:
        g[u].append((w, v))
        g[v].append((w, u))

    used, total = {0}, 0
    heap = [(w, v) for w, v in g[0]]
    import heapq
    heapq.heapify(heap)

    while heap and len(used) < n:
        w, v = heapq.heappop(heap)
        if v in used:
            continue
        used.add(v)
        total += w
        for cost, u in g[v]:
            if u not in used:
                heapq.heappush(heap, (cost, u))

    return total if len(used) == n else -1

print(prim(4, [(0,1,1),(0,2,4),(1,2,2),
               (1,3,5),(2,3,3)]))

def kruskal(n, edges):
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    total = count = 0
    for u, v, w in sorted(edges, key=lambda e: e[2]):
        a, b = find(u), find(v)
        if a != b:
            parent[a] = b
            total += w
            count += 1

    return total if count == n-1 else -1

print(kruskal(4, [(0,1,1),(0,2,4),(1,2,2),
                  (1,3,5),(2,3,3)]))

import heapq

def dijkstra(n, edges, src=0):
    g = [[] for _ in range(n)]
    for u, v, w in edges:
        g[u].append((v, w))
        g[v].append((u, w))

    d = [float('inf')]*n
    d[src] = 0
    q = [(0, src)]

    while q:
        dist, u = heapq.heappop(q)
        if dist != d[u]:
            continue
        for v, w in g[u]:
            if dist+w < d[v]:
                d[v] = dist+w
                heapq.heappush(q, (d[v], v))
    return d

print(dijkstra(4, [(0,1,1),(0,2,4),(1,2,2),
                   (1,3,5),(2,3,3)]))

def fractional(items, cap):
    items.sort(key=lambda x: x[1]/x[0], reverse=True)
    total = 0
    for w, v in items:
        take = min(w, cap)
        total += take * v/w
        cap -= take
        if cap == 0:
            break
    return total

print(fractional([(10,60),(20,100),(30,120)], 50))

def bellman(n, edges, src=0):
    d = [float('inf')]*n
    d[src] = 0

    for _ in range(n-1):
        changed = False
        for u, v, w in edges:
            if d[u] != float('inf') and d[u]+w < d[v]:
                d[v] = d[u]+w
                changed = True
        if not changed:
            break

    for u, v, w in edges:
        if d[u] != float('inf') and d[u]+w < d[v]:
            return "Negative cycle"

    return d

print(bellman(4, [(0,1,4),(0,2,5),(1,2,-2),
                  (2,3,3)]))
